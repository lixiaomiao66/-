    <script>
        // 1. 夸张的加载过程（8秒）
        const logs = [
            "Connecting to local database...",
            "Loading user profile...",
            "Analyzing jump rope frequency...",
            "Calculating fatigue index...",
            "Verifying penalty rules...",
            "Applying 'No-Penalty' protection...",
            "System ready."
        ];
        
        let progress = 0;
        const fill = document.getElementById('progressFill');
        const logEl = document.getElementById('logText');
        const loader = document.getElementById('system-loader');
        const mainPanel = document.getElementById('main-panel');

        const loadInterval = setInterval(() => {
            progress += Math.random() * 15;
            if (progress > 100) progress = 100;
            fill.style.width = progress + '%';
            
            const logIndex = Math.min(Math.floor(progress / 15), logs.length - 1);
            logEl.textContent = logs[logIndex];

            if (progress >= 100) {
                clearInterval(loadInterval);
                setTimeout(() => {
                    loader.style.opacity = 0;
                    setTimeout(() => {
                        loader.style.display = 'none';
                        mainPanel.style.display = 'block';
                    }, 1000);
                }, 500);
            }
        }, 200);

        // 2. 核心逻辑：滑块怎么拖都是 0
        const slider = document.getElementById('priceSlider');
        const priceText = document.getElementById('priceText');

        slider.addEventListener('input', function() {
            // 强制归零
            this.value = 0;
            priceText.textContent = '¥ 0.00';
            
            // 给一点物理反馈，让他觉得“卡住了”
            priceText.style.transform = 'scale(1.05)';
            setTimeout(() => { priceText.style.transform = 'scale(1)'; }, 100);
        });
